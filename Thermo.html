const wb = new ExcelJS.Workbook(); // ลบ Worksheet ทั้งหมดที่มีอยู่ wb.worksheets.forEach(sheet => { wb.removeWorksheet(sheet.id); }); // สร้างเพียง 1 Worksheet เท่านั้น const ws = wb.addWorksheet("Thermography Report"); // รวมข้อมูลทุกส่วนไว้ใน ws ตัวเดียว // ห้ามใช้ wb.addWorksheet() อีกในส่วนอื่น ws.columns = [ { width: 22 }, { width: 16 }, { width: 14 }, { width: 14 }, { width: 40 } ]; // ตั้งค่ากระดาษ A4 หน้าเดียว ws.pageSetup = { paperSize: 9, orientation: "portrait", fitToPage: true, fitToWidth: 1, fitToHeight: 1 }; // กำหนดพื้นที่พิมพ์ใน Sheet เดียว ws.pageSetup.printArea = "A1:E35"; // ตรวจสอบว่ามีเพียง 1 Sheet console.log("จำนวน Sheet:", wb.worksheets.length); // ต้องได้ 1 เท่านั้น if (wb.worksheets.length !== 1) { throw new Error("พบ Worksheet มากกว่า 1 Sheet"); }
